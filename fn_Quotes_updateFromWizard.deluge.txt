string automation.fn_Quotes_updateFromWizard(String quoteId, String payload)
{
	// v1.0 | 11/11/2026 | David Domingues
	// ref. MR-XXX (a definir ainda)
	// Revisão do Pedido de Orçamento (Quotes) pelo widget: atualiza o orçamento pela API v8 e SUBSTITUI as linhas de Quoted_Items.
	// Só permite rever nas fases Rascunho e Revisão (BP02).
	// Parâmetros: quoteId = ID do orçamento; payload = JSON montado pelo widget (campos do Quotes + Quoted_Items).
	DEBUG = false;
	CRM_API = "https://www.zohoapis.eu/crm/v8/";
	FASES_EDITAVEIS = {"Rascunho","Revisão"};
	resultado = "";
	info "INÍCIO fn_Quotes_updateFromWizard | quoteId=" + quoteId;
	try
	{
		// 1) Ler o orçamento atual: fase e IDs das linhas existentes
		atual = invokeurl
		[
			url :CRM_API + "Quotes/" + quoteId + "?fields=Quote_Stage,Quoted_Items"
			type :GET
			connection:"crm_quotes"
		];
		if(atual.get("data") == null || atual.get("data").isEmpty())
		{
			info "ERRO: orçamento não encontrado: " + atual.toString();
			item = Map();
			item.put("code","ERROR");
			item.put("message","Orçamento não encontrado.");
			lista = List();
			lista.add(item);
			erro = Map();
			erro.put("data",lista);
			resultado = erro.toString();
		}
		else
		{
			orcamento = atual.get("data").get(0);
			fase = ifnull(orcamento.get("Quote_Stage"),"");
			if(!FASES_EDITAVEIS.contains(fase))
			{
				// Segurança no servidor: o widget já bloqueia, mas a função não confia nisso
				info "AVISO: fase não editável: " + fase;
				item = Map();
				item.put("code","ERROR");
				item.put("message","O orçamento está na fase " + fase + " e não pode ser revisto.");
				lista = List();
				lista.add(item);
				erro = Map();
				erro.put("data",lista);
				resultado = erro.toString();
			}
			else
			{
				// 2) Montar as linhas: as existentes são apagadas (_delete) e as do widget entram como novas
				dataMap = payload.toMap();
				linhasFinais = List();
				for each  antiga in ifnull(orcamento.get("Quoted_Items"),List())
				{
					apagar = Map();
					apagar.put("id",antiga.get("id"));
					apagar.put("_delete",null);
					linhasFinais.add(apagar);
				}
				for each  nova in ifnull(dataMap.get("Quoted_Items"),List())
				{
					linhasFinais.add(nova);
				}
				dataMap.put("Quoted_Items",linhasFinais);
				listaData = List();
				listaData.add(dataMap);
				gatilhos = List();
				gatilhos.add("workflow");
				corpo = Map();
				corpo.put("data",listaData);
				corpo.put("trigger",gatilhos);
				if(DEBUG)
				{
					info "Corpo enviado: " + corpo.toString();
				}
				// 3) Atualizar (o nome da ligação tem de ser escrito diretamente)
				resp = invokeurl
				[
					url :CRM_API + "Quotes/" + quoteId
					type :PUT
					parameters:corpo.toString()
					connection:"crm_quotes"
				];
				if(DEBUG)
				{
					info "Resposta: " + resp.toString();
				}
				dados = resp.get("data");
				if(dados == null || dados.isEmpty())
				{
					info "ERRO: resposta sem data: " + resp.toString();
				}
				else
				{
					info "FIM fn_Quotes_updateFromWizard | " + dados.get(0).get("code");
				}
				resultado = resp.toString();
			}
		}
	}
	catch (e)
	{
		info "ERRO fn_Quotes_updateFromWizard: " + e.toString();
		item = Map();
		item.put("code","ERROR");
		item.put("message",e.toString());
		lista = List();
		lista.add(item);
		erro = Map();
		erro.put("data",lista);
		resultado = erro.toString();
	}
	return resultado;
}
