string automation.fn_Quotes_createFromWizard(String payload)
{
	// v1.2 | 11/11/2026 | David Domingues
	// ref. MR-XXX (a definir ainda)
	// Cria o Pedido de Orçamento (Quotes) pela API v8 com o JSON montado pelo widget (campos do Quotes + Quoted_Items com as colunas personalizadas das linhas).
	DEBUG = false;
	CRM_API = "https://www.zohoapis.eu/crm/v8/";
	resultado = "";
	info "INÍCIO fn_Quotes_createFromWizard";
	try
	{
		// O widget envia o orçamento completo em texto JSON
		dataMap = payload.toMap();
		listaData = List();
		listaData.add(dataMap);
		gatilhos = List();
		gatilhos.add("workflow");
		corpo = Map();
		corpo.put("data",listaData);
		corpo.put("trigger",gatilhos);
		if(DEBUG)
		{
			info "Corpo enviado: " + corpo.toString();
		}
		// O nome da ligação tem de ser escrito diretamente (não aceita variável)
		resp = invokeurl
		[
			url :CRM_API + "Quotes"
			type :POST
			parameters:corpo.toString()
			connection:"crm_quotes"
		];
		if(DEBUG)
		{
			info "Resposta: " + resp.toString();
		}
		// Validar a resposta: tem de trazer data com pelo menos um registo
		dados = resp.get("data");
		if(dados == null || dados.isEmpty())
		{
			info "ERRO: resposta sem data: " + resp.toString();
		}
		else
		{
			info "FIM fn_Quotes_createFromWizard | " + dados.get(0).get("code");
		}
		resultado = resp.toString();
	}
	catch (e)
	{
		info "ERRO fn_Quotes_createFromWizard: " + e.toString();
		item = Map();
		item.put("code","ERROR");
		item.put("message",e.toString());
		lista = List();
		lista.add(item);
		erro = Map();
		erro.put("data",lista);
		resultado = erro.toString();
	}
	return resultado;
}
